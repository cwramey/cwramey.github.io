shortest_route.py
import geopandas as gpd
import networkx as nx
from shapely.geometry import LineString

roads = gpd.read_file("roads.geojson")

# Build a weighted road graph
G = nx.Graph()
for _, row in roads.iterrows():
    start, end = row.geometry.coords[0], row.geometry.coords[-1]
    G.add_edge(start, end, weight=row.geometry.length)

# Find and measure the shortest route
path = nx.shortest_path(G, source=A, target=B, weight="weight")
distance = sum(G[u][v]["weight"] for u, v in zip(path[:-1], path[1:]))

print("Shortest route:", path)
print("Total distance:", round(distance, 2), "meters")

route = gpd.GeoDataFrame(geometry=[LineString(path)], crs=roads.crs)
route.to_file("shortest_route.geojson", driver="GeoJSON")
